cmake_minimum_required(VERSION 3.25)
project(PSVR2SimShaker VERSION 0.3.0 LANGUAGES CXX)
set(PSVR2SIMSHAKER_VERSION "${PROJECT_VERSION}-alpha.3")
set(PSVR2SIMSHAKER_DISPLAY_VERSION "0.3.0 Alpha 3")
configure_file(src/version.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/version.hpp" @ONLY)
if(NOT WIN32)
  message(FATAL_ERROR "PSVR2SimShaker currently targets Windows x64.")
endif()
enable_language(RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/permissive-;/EHsc>")
find_package(imgui CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_library(shaker_core src/core.cpp src/haptic_tests.cpp)
target_include_directories(shaker_core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(shaker_core PUBLIC nlohmann_json::nlohmann_json)
add_library(shaker_platform src/platform.cpp src/telemetry.cpp src/integration.cpp src/haptics.cpp src/updates.cpp)
target_link_libraries(shaker_platform PUBLIC shaker_core shell32 shlwapi advapi32 ole32 bcrypt winhttp setupapi)
add_library(PSVR2SimShakerDcsBridge SHARED src/dcs_bridge.cpp src/telemetry.cpp)
target_include_directories(PSVR2SimShakerDcsBridge PRIVATE src)
set_target_properties(PSVR2SimShakerDcsBridge PROPERTIES PREFIX "" OUTPUT_NAME "PSVR2SimShakerDcsBridge")
configure_file(src/resources.rc.in "${CMAKE_CURRENT_BINARY_DIR}/resources.rc" @ONLY)
add_executable(PSVR2SimShaker WIN32 src/main.cpp src/app.cpp src/logo.cpp "${CMAKE_CURRENT_BINARY_DIR}/resources.rc")
target_link_libraries(PSVR2SimShaker PRIVATE shaker_platform imgui::imgui d3d11 dxgi d3dcompiler comdlg32 windowscodecs)
add_dependencies(PSVR2SimShaker PSVR2SimShakerDcsBridge)
add_custom_command(TARGET PSVR2SimShaker POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/dcs" "$<TARGET_FILE_DIR:PSVR2SimShaker>/dcs"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:PSVR2SimShaker>/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/compatibility.json" "$<TARGET_FILE_DIR:PSVR2SimShaker>/compatibility.json"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/LICENSE" "$<TARGET_FILE_DIR:PSVR2SimShaker>/LICENSE")
enable_testing()
add_executable(shaker_tests tests/tests.cpp)
target_link_libraries(shaker_tests PRIVATE shaker_platform)
add_test(NAME shaker_tests COMMAND shaker_tests)
add_executable(haptic_pattern_tests tests/haptic_pattern_tests.cpp)
target_link_libraries(haptic_pattern_tests PRIVATE shaker_core)
add_test(NAME haptic_pattern_tests COMMAND haptic_pattern_tests)
add_executable(aircraft_tests tests/aircraft_tests.cpp)
target_link_libraries(aircraft_tests PRIVATE shaker_core)
add_test(NAME aircraft_tests COMMAND aircraft_tests)
find_program(SHAKER_LUA NAMES luajit lua5.1)
if(SHAKER_LUA)
  add_test(NAME export_tests COMMAND "${SHAKER_LUA}" "${CMAKE_SOURCE_DIR}/tests/export_tests.lua" "${CMAKE_SOURCE_DIR}/dcs/Export.lua")
endif()
add_executable(update_tests tests/update_tests.cpp)
target_link_libraries(update_tests PRIVATE shaker_platform)
add_test(NAME update_tests COMMAND update_tests)
add_executable(bridge_tests tests/bridge_tests.cpp)
target_link_libraries(bridge_tests PRIVATE shaker_platform)
add_dependencies(bridge_tests PSVR2SimShakerDcsBridge)
add_test(NAME bridge_tests COMMAND bridge_tests "$<TARGET_FILE:PSVR2SimShakerDcsBridge>")
add_library(fake_toolkit SHARED tests/fake_toolkit.cpp)
add_executable(host_tests tests/host_tests.cpp)
target_link_libraries(host_tests PRIVATE shaker_platform)
add_dependencies(host_tests fake_toolkit)
set_target_properties(fake_toolkit host_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/host-tests/$<CONFIG>")
add_test(NAME host_tests COMMAND host_tests)
install(TARGETS PSVR2SimShaker PSVR2SimShakerDcsBridge RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(DIRECTORY dcs licenses DESTINATION .)
install(FILES docs/AIRCRAFT.md docs/EFFECTS.md docs/FUTURE_WORK.md docs/HARDWARE.md docs/PROTOCOL.md docs/SETUP.md docs/TELEMETRY.md docs/TESTING.md DESTINATION docs OPTIONAL)
install(FILES assets/PSVR2SimShaker.ico assets/PSVR2SimShaker.png assets/README.md DESTINATION assets)
install(FILES README.md CHANGELOG.md LICENSE THIRD_PARTY_NOTICES.md compatibility.json DESTINATION .)
